// ADD inside your EXISTING /databases/{database}/documents block.
// Keep ALL existing rules, including interruptions. This snippet is NOT the entire rules file.
match /demandRecords/{equipmentId} {
  function demandEditor() {
    return request.auth != null
      && exists(/databases/$(database)/documents/allowedUsers/$(request.auth.uid))
      && get(/databases/$(database)/documents/allowedUsers/$(request.auth.uid)).data.enabled == true
      && get(/databases/$(database)/documents/allowedUsers/$(request.auth.uid)).data.role in ['admin','editor'];
  }
  function demandMetric(v) {
    return v == null || (v is map
      && v.keys().hasOnly(['mw','date','time']) && v.keys().hasAll(['mw','date','time'])
      && v.mw is number && v.mw >= 0
      && v.date is string && v.date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
      && v.time is string && (v.time == '' || v.time.matches('^([01][0-9]|2[0-3]):[0-5][0-9]$')));
  }
  function validDemand() {
    let d = request.resource.data;
    let names = {'rtpp1':'RTPP-1','rtpp2':'RTPP-2','chittoor1':'Chittoor-1','chittoor2':'Chittoor-2','ict1':'ICT-1','ict2':'ICT-2','kalikiri1':'Kalikiri-1','kalikiri2':'Kalikiri-2','madanapalli1':'Madanapalli-1','madanapalli2':'Madanapalli-2'};
    return equipmentId in names
      && d.keys().hasOnly(['equipment','max','min','version','savedBy','savedAt'])
      && d.keys().hasAll(['equipment','max','min','version','savedBy','savedAt'])
      && d.equipment == names[equipmentId]
      && demandMetric(d.max) && demandMetric(d.min)
      && (d.max == null || d.min == null || d.min.mw <= d.max.mw)
      && d.version is int && d.version > 0
      && d.savedBy == request.auth.uid
      && d.savedAt is string && d.savedAt.size() <= 30;
  }
  allow read: if request.auth != null;
  allow create: if demandEditor() && validDemand() && request.resource.data.version == 1;
  allow update: if demandEditor() && validDemand() && request.resource.data.version == resource.data.version + 1;
  allow delete: if false;
}
