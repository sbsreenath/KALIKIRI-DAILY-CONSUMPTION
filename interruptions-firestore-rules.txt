match /interruptions/{entryId} {
  function interruptionEditor() {
    return request.auth != null
      && exists(/databases/$(database)/documents/allowedUsers/$(request.auth.uid))
      && get(/databases/$(database)/documents/allowedUsers/$(request.auth.uid)).data.enabled == true
      && get(/databases/$(database)/documents/allowedUsers/$(request.auth.uid)).data.role in ['admin', 'editor'];
  }
  function validInterruption() {
    let d = request.resource.data;
    return d.keys().hasOnly(['equipment','from','to','reason','mode','sourceKey','sourceDate','savedBy','savedAt'])
      && d.keys().hasAll(['equipment','from','to','reason','mode','sourceKey','sourceDate','savedBy','savedAt'])
      && d.equipment in ['RTPP-1','RTPP-2','Chittoor-1','Chittoor-2','Kalikiri-1','Kalikiri-2','Madanapalli-1','Madanapalli-2','ICT-1','ICT-2']
      && d.from is string && d.to is string
      && d.from.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}$')
      && d.to.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}$')
      && d.to >= d.from
      && d.reason is string && d.reason.size() > 0 && d.reason.size() <= 10000
      && d.mode in ['manual','correction']
      && d.sourceKey is string && d.sourceKey.size() <= 200
      && d.sourceDate is string && d.sourceDate.size() <= 10
      && (d.mode != 'correction' || d.sourceKey.size() > 0)
      && d.savedBy == request.auth.uid
      && d.savedAt is string && d.savedAt.size() <= 30;
  }
  allow read: if request.auth != null;
  allow create, update: if interruptionEditor() && validInterruption();
  allow delete: if interruptionEditor();
}
